import streamlit as st

# 1. ページの基本設定
st.set_page_config(
    page_title="京都かわ瀬 弁当システム",
    page_icon="🍱",
    layout="centered"
)

# 2. アクセス制限（OS・スペック）を行うJavaScriptを埋め込む
# ※ブラウザ側で判定し、条件を満たしていない場合はエラー画面に書き換える
st.markdown("""
<script>
// CPUコア数とRAMの取得（大まかな目安）
const cpuCores = navigator.hardwareConcurrency || 1;
const deviceRAM = navigator.deviceMemory || 1;

const MIN_CORES = 4;
const MIN_RAM = 8;
const isSpecPassed = (cpuCores >= MIN_CORES) && (deviceRAM >= MIN_RAM);

// OSの判定
const ua = navigator.userAgent;
let isAllowedOS = false;

// Windows 10 または 11 の判定
if (ua.indexOf("Windows NT 10.0") !== -1) {
    isAllowedOS = true;
}

// Androidのバージョン判定 (14, 13, 11, 10)
const androidMatch = ua.match(/Android\s([0-9\.]+)/);
if (androidMatch) {
    const androidVersion = parseInt(androidMatch[1], 10);
    if ([14, 13, 11, 10].includes(androidVersion)) {
        isAllowedOS = true;
    }
}

// 条件を満たしていない場合、強制的に画面をブロックする
if (!isAllowedOS || !isSpecPassed) {
    document.addEventListener("DOMContentLoaded", function() {
        document.body.innerHTML = `
            <div style="display: flex; justify-content: center; align-items: center; height: 100vh; background-color: #f8d7da; color: #721c24; font-family: sans-serif; text-align: center; padding: 20px;">
                <div>
                    <h1 style="font-size: 24px; margin-bottom: 10px;">⚠️ アクセスエラー（利用環境外）</h1>
                    <p style="font-size: 16px;">このアプリは指定された対応OS（Windows 10/11、Android 14/13/11/10）および<br>最低スペック（4コア以上・RAM 8GB以上）の端末からのみ利用できます。</p>
                </div>
            </div>
        `;
    });
}
</script>
""", unsafe_allow_html=True)

# 3. スペック・OS条件をクリアした人が見るアプリ本編
st.title("🍱 京都かわ瀬 弁当注文・集計システム")
st.write("ようこそ！本日の弁当の注文・メニュー管理を行います。")

# 注文フォームのサンプル
st.subheader("本日のメニュー選択")
bento_type = st.selectbox(
    "お好みの弁当を選んでください",
    ["特製幕の内弁当", "唐揚げマヨ弁当", "日替わり魚弁当", "ヘルシーサラダ弁当"]
)

if st.button("注文を確定する"):
    st.success(f"「{bento_type}」の注文を受け付けました！")
